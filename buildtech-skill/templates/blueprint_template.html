1. Product Overview & System Architecture

The system continuously monitors soil volumetric water content and ambient temperature. When soil moisture drops below 35%, the system opens a 5V solenoid water valve for 15 seconds, displays status diagnostics on an I2C LCD screen, and blinks a warning indicator. All telemetry is broadcast over serial and Wi-Fi.

5. Production Firmware (C++ / Arduino IDE)

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define PIN_MOISTURE   34
#define PIN_RELAY      18
#define PIN_ALERT_LED  4
#define PIN_STATUS_LED 2

#define DRY_THRESHOLD  35   // Percentage below which watering triggers
#define WATER_DURATION 5000 // 5 seconds pump run

LiquidCrystal_I2C lcd(0x27, 16, 2);
unsigned long lastSampleTime = 0;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RELAY, OUTPUT);
  pinMode(PIN_ALERT_LED, OUTPUT);
  pinMode(PIN_STATUS_LED, OUTPUT);

  // Relay is Active-LOW
  digitalWrite(PIN_RELAY, HIGH);
  digitalWrite(PIN_ALERT_LED, LOW);

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("BuildTech Plant");
  lcd.setCursor(0, 1);
  lcd.print("System Ready...");
  delay(1500);
}

void loop() {
  unsigned long now = millis();
  if (now - lastSampleTime >= 2000) {
    lastSampleTime = now;

    int rawAdc = analogRead(PIN_MOISTURE);
    // ESP32 12-bit ADC: 0 to 4095
    int moisturePercent = map(rawAdc, 4095, 1500, 0, 100);
    moisturePercent = constrain(moisturePercent, 0, 100);

    Serial.printf("[TELEMETRY] Moisture: %d%% (ADC: %d)\n", moisturePercent, rawAdc);

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.printf("Moisture: %d%%", moisturePercent);

    if (moisturePercent < DRY_THRESHOLD) {
      lcd.setCursor(0, 1);
      lcd.print("STATUS: PUMP ON");
      digitalWrite(PIN_ALERT_LED, HIGH);
      digitalWrite(PIN_RELAY, LOW); // Trigger pump
      delay(WATER_DURATION);
      digitalWrite(PIN_RELAY, HIGH);
      digitalWrite(PIN_ALERT_LED, LOW);
    } else {
      lcd.setCursor(0, 1);
      lcd.print("STATUS: OPTIMAL");
    }
  }
}

6. Assembly Instructions & First-Boot Checklist

  1. Step 1 (Power Bus): Connect the USB-C / 5V power supply. Never connect 5V directly to ESP32 GPIO pins.
  2. Step 2 (Ground Rail): Ensure all modules share a unified common GND rail.
  3. Step 3 (I2C Scanning): Flash an I2C scanner sketch to verify LCD address (typically 0x27 or 0x3F).
  4. Step 4 (Calibration): Calibrate sensor readings in completely dry soil (0%) and water-saturated soil (100%).
